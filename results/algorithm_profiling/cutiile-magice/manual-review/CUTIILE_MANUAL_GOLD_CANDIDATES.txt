using namespace std;

class Task {
 public:
    void solve() {
        read_input();
        get_result();
    }

 private:
    double n, m;

    void read_input() {
        cin >> n >> m;
    }

    void get_result() {
        double res = 0;
        vector<vector<double>> dp(m, vector<double> (n + 1, 0));
        
        
        if (m == 0) {
            cout << n << '\n';
            return;
        }
        for (int i = 0; i < n; ++i) {
            dp[0][i] = (double)(1/n);
        }
        
        
        for (int i = 1; i < m; ++i) {
            for (int j = n - 1; j >= 0; --j) {
                dp[i][j] = dp[i][j + 1] + dp[i - 1][j] / (double)(j + 1);
            }
        }
        
        
        for (int i = 0; i < n; ++i) {
            res += ((double)(i + 1) * dp[m - 1][i]);
        }
        
        printf("%.9lf\n", res);
    }

};

    Task *task = new Task();
    task->solve();
    delete task;
    return 0;

double prob(double n, int k) {
    if (k == 0) {
        return n;
    }
    double sum = 0;
    for(int i = 1; i <= n; i++) {
          sum += (prob(i, k - 1));
    }
    return (1/n) * sum;
}
