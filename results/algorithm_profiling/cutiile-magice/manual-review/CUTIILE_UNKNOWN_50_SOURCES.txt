import java.io.*;
import java.util.*;
import java.text.*;
import java.math.*;
import java.util.regex.*;

class MyScanner {
    BufferedReader br;
    StringTokenizer st;
 
    public MyScanner() {
        br = new BufferedReader(new InputStreamReader(System.in));
    }
 
    String next() {
        while (st == null || !st.hasMoreElements()) {
            try {
                st = new StringTokenizer(br.readLine());
            } catch (IOException e) {
                e.printStackTrace();
            }
        }
        return st.nextToken();
    }
 
    int nextInt() {
        return Integer.parseInt(next());
    }
 
    long nextLong() {
        return Long.parseLong(next());
    }
 
    double nextDouble() {
        return Double.parseDouble(next());
    }
 
    String nextLine(){
        String str = "";
        try {
            str = br.readLine();
        } catch (IOException e) {
            e.printStackTrace();
        }
        return str;
    }
}

public class Solution {
    static class Task {
        int n;
        int k;

        private void readInput() {
            MyScanner sc = new MyScanner();
            n = sc.nextInt();
            k = sc.nextInt();
        }

        private void writeOutput(double result) {
            System.out.format("%.9f", result);
        }
        
        private double getResult() {
            
            double result = 0;
            
            if(k == 0) {
                return n;
            }
            
            double baseProbability = 1 / (double)n;
            
            double [][] probs = new double[n+1][k+1];
            
            for(int i = 1; i <= n; i++){
                probs[i][1] = baseProbability;
            }
            for(int i = 2; i <= k; i++){
                probs[n][i] = probs[n][i-1] * baseProbability;
            }
            
            for(int j = 2; j <= k; j++){
                for(int i = n-1; i >= 1; i--){
                    double currProb = 1 / (double) i;
                    probs[i][j] = probs[i][j-1] * currProb + probs[i+1][j];
                }
            }
            
            for(int i = 1; i <= n; i++) {
                result += probs[i][k] * i;
            }
            
            return result;
        }

        public void solve() {
            readInput();
            writeOutput(getResult());
        }
    }
    
    public static void main(String[] args) {
        new Task().solve();
    }
}

    }

int main() {
    int n;
    int k;
    cin >> n >> k;
    vector<vector<double>> dp(k, vector<double> (n + 1, 0));
    
    if (k == 0) {
       cout << setprecision(9) << (double)n; 
       return 0;
    }
    for (int i = 0; i < n; i++) {
        dp[0][i]  = (double)(1.0 / n);
    }
    for (int i = 1; i < k; i++) {
        for (int j = n - 1; j >= 0; j--) {
            dp[i][j] = dp[i][j + 1] + dp[i - 1][j] /(double)(j + 1);
        }
    }
    double percentage = 0;
    for (int i = 0; i < n; i++) {
        percentage += ((double)(i+1) * dp[k - 1][i]);
    }
    cout << setprecision(9) << percentage;
    return 0;
}
