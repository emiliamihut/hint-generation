#include <iostream>
#include <vector>
#include <limits>
using namespace std;
#define INF numeric_limits<int>::max()

class Solution {
    vector<vector<int>> diamonds;
    int n, m;
    int max_moves;
    pair<int, int> start;
    int max_diamond_value = -INF;

public:
    void read() {
        cin >> n >> m >> max_moves;
        cin >> start.first >> start.second;

        diamonds.resize(n + 1, vector<int>(m + 1));
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                cin >> diamonds[i][j];
            }
        }
    }

    void solve() {
        
        vector<vector<vector<int>>> dp(n + 1, vector<vector<int>>(m + 1, vector<int>(max_moves + 1, -INF)));
        dp[start.first][start.second][0] = diamonds[start.first][start.second];

        
        for (int k = 1; k <= max_moves; ++k) {
            for (int i = 1; i <= n; ++i) {
                for (int j = 1; j <= m; ++j) {
                    int max_from_neighbors = -INF;

                    
                    if (i > 1) max_from_neighbors = max(max_from_neighbors, dp[i-1][j][k-1]);
                    if (i < n) max_from_neighbors = max(max_from_neighbors, dp[i+1][j][k-1]);
                    if (j > 1) max_from_neighbors = max(max_from_neighbors, dp[i][j-1][k-1]);
                    if (j < m) max_from_neighbors = max(max_from_neighbors, dp[i][j+1][k-1]);

                    dp[i][j][k] = max(dp[i][j][k-1], max_from_neighbors);
                    if (dp[i][j][k] != -INF) dp[i][j][k] = max(dp[i][j][k], diamonds[i][j]);
                }
            }
        }

        for (int i = 1; i <= n; ++i) {
            for (int j = 1; j <= m; ++j) {
                max_diamond_value = max(max_diamond_value, dp[i][j][max_moves]);
            }
        }
    }

    void print() {
        cout << max_diamond_value << endl;
    }
};

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    Solution solution;
    solution.read();
    solution.solve();
    solution.print();
    return 0;
}

#include <iostream>
#include <vector>
#include <queue>
#include <algorithm>
using namespace std;

int main() {
    int n, m, p, l, c;
    cin >> n >> m >> p >> l >> c;

    vector<vector<int>> mat(n + 1, vector<int>(m + 1));
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j <= m; ++j)
            cin >> mat[i][j];

    vector<vector<vector<bool>>> dp(p + 1, vector<vector<bool>>(n + 1, vector<bool>(m + 1, false)));
    dp[0][l][c] = true;

    int di[] = {-1, 1, 0, 0};
    int dj[] = {0, 0, -1, 1};

    for (int step = 1; step <= p; ++step) {
        for (int i = 1; i <= n; ++i) {
            for (int j = 1; j <= m; ++j) {
                if (!dp[step - 1][i][j]) continue;
                for (int d = 0; d < 4; ++d) {
                    int ni = i + di[d];
                    int nj = j + dj[d];
                    if (ni >= 1 && ni <= n && nj >= 1 && nj <= m)
                        dp[step][ni][nj] = true;
                }
            }
        }
    }

    int ans = -1e9;
    for (int i = 1; i <= n; ++i)
        for (int j = 1; j <= m; ++j)
            if (dp[p][i][j])
                ans = max(ans, mat[i][j]);

    cout << ans << "\n";
    return 0;
}
