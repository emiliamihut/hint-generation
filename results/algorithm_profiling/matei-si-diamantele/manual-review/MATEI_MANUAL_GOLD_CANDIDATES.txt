import java.io.*;
import java.util.*;

public class Solution {

    static class Point {
        int x, y, steps;
        public Point(int x, int y, int steps) {
            this.x = x;
            this.y = y;
            this.steps = steps;
        }
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int m = sc.nextInt();
        int p = sc.nextInt();
        int l = sc.nextInt() - 1;
        int c = sc.nextInt() - 1;

        int[][] grid = new int[n][m];
        for (int i = 0; i < n; i++)
            for (int j = 0; j < m; j++)
                grid[i][j] = sc.nextInt();

        boolean[][] visited = new boolean[n][m];
        Queue<Point> queue = new LinkedList<>();
        queue.add(new Point(l, c, 0));
        visited[l][c] = true;

        int maxBeauty = grid[l][c];

        int[] dx = {-1, 1, 0, 0};
        int[] dy = {0, 0, -1, 1};

        while (!queue.isEmpty()) {
            Point curr = queue.poll();

            if (curr.steps == p) continue;

            for (int d = 0; d < 4; d++) {
                int nx = curr.x + dx[d];
                int ny = curr.y + dy[d];

                if (nx >= 0 && nx < n && ny >= 0 && ny < m && !visited[nx][ny]) {
                    visited[nx][ny] = true;
                    maxBeauty = Math.max(maxBeauty, grid[nx][ny]);
                    queue.add(new Point(nx, ny, curr.steps + 1));
                }
            }
        }

        System.out.println(maxBeauty);
    }
}
