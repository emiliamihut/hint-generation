const int dx[] = {-1, 0, 1, 0};
const int dy[] = {0, 1, 0, -1};

int n, m, p, l, c;
vector<vector<int>> grid;
vector<vector<bool>> visited;

bool valid(int x, int y) {
    return x >= 0 && x < n && y >= 0 && y < m;
}

int dfs(int x, int y, int moves_left, int beauty) {
    if (moves_left == 0) {
        return beauty;
    }
    int max_beauty = beauty;
    for (int i = 0; i < 4; i++) {
        int nx = x + dx[i];
        int ny = y + dy[i];
        if (valid(nx, ny) && !visited[nx][ny]) {
            visited[nx][ny] = true;
            int new_beauty = grid[nx][ny];
            max_beauty = max(max_beauty, dfs(nx, ny, moves_left - 1, new_beauty));
            visited[nx][ny] = false;
        }
    }
    return max_beauty;
}

int main() {
    cin >> n >> m >> p >> l >> c;
    grid.resize(n, vector<int>(m));
    visited.resize(n, vector<bool>(m, false));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            cin >> grid[i][j];
        }
    }
    visited[l-1][c-1] = true;
    int max_beauty = dfs(l-1, c-1, p, grid[l-1][c-1]);
    cout << max_beauty << endl;
    return 0;
}

using namespace std;

int backtracking(int n, int m, int steps, int curr_row, int curr_col, int max_val, vector<vector<int>> &matrix) {
    if (steps == 0 && curr_row >= 0 && curr_row < n && curr_col >= 0 && curr_col < m) {
        max_val = max(max_val, matrix[curr_row][curr_col]);
        return max_val;
    }

    if (curr_row < 0 || curr_row >= n || curr_col < 0 || curr_col >= m)
        return max_val;

    max_val = max(max_val, matrix[curr_row][curr_col]);

    
    max_val = backtracking(n, m, steps - 1, curr_row - 1, curr_col, max_val, matrix);
    
    max_val = backtracking(n, m, steps - 1, curr_row + 1, curr_col, max_val, matrix);
    
    max_val = backtracking(n, m, steps - 1, curr_row, curr_col - 1, max_val, matrix);
    
    max_val = backtracking(n, m, steps - 1, curr_row, curr_col + 1, max_val, matrix);

    return max_val;
}

int main() {
    int n, m, steps, start_row, start_col;
    cin >> n >> m >> steps >> start_row >> start_col;

    vector<vector<int>> matrix(n, vector<int>(m));
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++)
            cin >> matrix[i][j];
    }

    
    start_col--;
    start_row--;

    int max_val = matrix[start_row][start_col];
    max_val = backtracking(n, m, steps, start_row, start_col, max_val, matrix);

    cout << max_val << endl;
    
    return 0;
}
