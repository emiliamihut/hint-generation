public class Solution {
    int n;
    String group_1;
    String group_2;

    private void readInput() {
        MyScanner sc = new MyScanner();
        n = sc.nextInt();
        group_1 = sc.nextLine();
        group_2 = sc.nextLine();
    }

    private void writeOutput(int result) {
        System.out.println(result);
    }
    private int getResult() {
        int a[] = new int[255];
        int b[] = new int[255]; 
        for(int i = 0; i < n; i++){
            a[group_1.codePointAt(i)]++;
            b[group_2.codePointAt(i)]++;
        }
        int sol = 0;
        int unmatch_1 = 0;
        int unmatch_2 = 0;
        int group_1_unmatched = 0;
        int group_2_unmatched = 0;

        for(int i = 97; i < 123; i++){
            int minim = a[i];
            unmatch_1 = 1;
            if(b[i] < minim){
                minim = b[i];
                unmatch_1 = 0;
            }
            sol += minim;
            if(unmatch_1 == 1){
                group_2_unmatched += (b[i] - a[i]);
            }
            else{
                group_1_unmatched += (a[i] - b[i]);    
            }
        }
        int nr_stars_1 = a[42];
        int nr_stars_2 = b[42];


        if(nr_stars_1 > group_2_unmatched){
            sol += group_2_unmatched;
            nr_stars_1 -= group_2_unmatched;
        }
        else{
            sol += nr_stars_1;
            nr_stars_1 = 0;
        }

        if(nr_stars_2 > group_1_unmatched){
            sol += group_1_unmatched;
            nr_stars_2 -= group_1_unmatched;
        }
        else{
            sol += nr_stars_2;
            nr_stars_2 = 0;
        }

        if(nr_stars_1 != 0 && nr_stars_2 !=0){
            int minimum = nr_stars_1;
            if(minimum > nr_stars_2){
                minimum = nr_stars_2;
            }
            sol += minimum;
        }

        return sol;
    }
