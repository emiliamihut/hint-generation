#include <bits/stdc++.h>

class Task {
public:
    void solve() {
        read_input();
        compute_solution();
        print_output();
    }

private:
    int n;
    vector<int> sosete;
    int perechi=0;

    void read_input() {
        cin>>n;
        char x;
        for(int i = 1; i <= 2*n; i++){
            cin>>x;
            sosete.push_back(x);
        }
    }

    void compute_solution() {
        int x=0,y=0;
        sort(sosete.begin(),sosete.end());
        for(int i = 0 ; i < 2*n-1; i++) {
            if(sosete[i] == sosete[i+1]){
                i += 1;
                perechi++;
            }else{
                if(sosete[i] == 42)
                    x++;
                else y++;
            }
                
        }
        if(x > y)
            perechi = perechi + y + (x-y)/2;
        else 
            perechi = perechi + x;

    }

    void print_output() {
        cout<<perechi;
    }
};
