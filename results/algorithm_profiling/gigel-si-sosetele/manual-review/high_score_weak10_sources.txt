                indexRight++;
                continue;
            }
            
            if(left[indexLeft] == right[indexRight]) {
                pairs++;
                indexLeft++;
                indexRight++;
            } else {
                if(left[indexLeft] > right[indexRight]) {
                    if(universalMatchingLeft > 0) {
                        pairs++;
                        universalMatchingLeft--;
                        indexRight++;
                    } else {
                        indexRight++;
                    }
                } else {
                    if(universalMatchingRight > 0) {
                        pairs++;
                        universalMatchingRight--;
                        indexLeft++;
                    } else {
                        indexLeft++;
                    }
                }
            }
        }
        
        int remainingLeftSocks = left.length - indexLeft;
        int remainingRightSocks = right.length - indexRight;
        int newPairsLeft = Integer.min(remainingLeftSocks, universalMatchingRight);
        int newPairsRight = Integer.min(remainingRightSocks, universalMatchingLeft);
        pairs += (newPairsLeft + newPairsRight);
        universalMatchingLeft -= newPairsRight;
        universalMatchingRight -= newPairsLeft;
        if(universalMatchingLeft > 0 && universalMatchingRight > 0){
            pairs = pairs + Integer.min(universalMatchingLeft, universalMatchingRight);
        }
        return pairs;
    }
