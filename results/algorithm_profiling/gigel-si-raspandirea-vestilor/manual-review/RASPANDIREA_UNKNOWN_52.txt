        public int BFS(int source, int[] visited){
            int t = 0;
            ArrayList<Integer> q = new ArrayList<>();
            visited[source] = 1;
            q.add(source);

            while(!q.isEmpty()){
                source = q.remove(0);
                t++;
                ArrayList<Integer> neighbours = adj[source];
                for (int i = 0; i < neighbours.size(); i++){
                    int node = neighbours.get(i);
                    if(visited[node] == 0){
                        visited[node] = 1;
                        q.add(node);
                    }
                }
            }
            return t;
        }

                visited[i] = 0;
               r.adj[j-1].add(k);
           }

        int n,m;
        n = sc.nextInt();
        m = sc.nextInt();
        int[][] x = new int[n + 1][n + 1];
        
        boolean[] visited = new boolean[n + 1];
     
        for (int i = 0; i < m; i++) {
            int u = sc.nextInt();
            int v = sc.nextInt();
            x[u][v] = 1;
            x[v][u] = 1;
            
        }
        sc.close();
        int time = 0;
        for (int i = 1; i <= n; i++) {
            if (!visited[i]) {
                time++;
                visited[i] = true;
                for (int j = 1; j <= n; j++) {
                    if (x[i][j] == 1) {
                        visited[j] = true;
                    }
                }
            } else {
                continue;
            }
        }
        System.out.print(time);
    
    }
}
