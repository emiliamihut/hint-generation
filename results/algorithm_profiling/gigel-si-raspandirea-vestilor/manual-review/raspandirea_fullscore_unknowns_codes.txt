class MyScanner {
    BufferedReader br;
    StringTokenizer st;
 
    public MyScanner() {
        br = new BufferedReader(new InputStreamReader(System.in));
    }
 
    String next() {
        while (st == null || !st.hasMoreElements()) {
            try {
                st = new StringTokenizer(br.readLine());
            } catch (IOException e) {
                e.printStackTrace();
            }
        }
        return st.nextToken();
    }
 
    int nextInt() {
        return Integer.parseInt(next());
    }
 
    long nextLong() {
        return Long.parseLong(next());
    }
 
    double nextDouble() {
        return Double.parseDouble(next());
    }
 
    String nextLine(){
        String str = "";
        try {
            str = br.readLine();
        } catch (IOException e) {
            e.printStackTrace();
        }
        return str;
    }
}

public class Solution {
    
    public static int INF = (int)1e9;

    public static void main(String[] args) {
        
        MyScanner sc = new MyScanner();
        int N = sc.nextInt();
        int M = sc.nextInt();
        
        ArrayList<Integer> adj[] = new ArrayList[N + 1];
        
        for (int node = 0; node <= N; node++) {
            adj[node] = new ArrayList<>();
        }
        
        for (int i = 0; i < M; i++) {
            int x = sc.nextInt();
            int y = sc.nextInt();
            adj[x].add(y);
            adj[y].add(x);
        }
        
        Integer d[] = new Integer[N + 1];
        Arrays.fill(d, INF);
        d[0] = 0;
        d[1] = 0;
        
        Queue<Integer> q = new LinkedList<>();
        q.add(1);
        
        while(!q.isEmpty()) {
            int node = q.poll();
            for (Integer neigh: adj[node]) {
                if (d[node] + 1 < d[neigh]) {
                    d[neigh] = d[node] + 1;
                    q.add(neigh);
                }
            }
        }
        
        
        for (int i = 1; i <= N; i++) {
            if (d[i] == INF) {
                d[i] = -1;
            }
        }
        
        Arrays.sort(d, Collections.reverseOrder());
        
        System.out.println(d[0]);
    }
}

int bfs(vector<vector<int>>& graph) {
    int start = 1;
    int timestamp = 0;
    queue<int> q;
    q.push(start);
    vector<int> visited(graph.size(), -1);
    visited[0] = 0;
    while (!q.empty()) {
        int node = q.front();
        if (visited[node - 1] != timestamp)
            timestamp++;
        q.pop();
        for (int i = 0; i < graph[node - 1].size(); i++) {
            if (visited[graph[node - 1][i] - 1] == -1) {
                q.push(graph[node - 1][i]);
                visited[graph[node - 1][i] - 1] = timestamp + 1;
            }
        }
        
    }
    return timestamp;
    
}

void dfs(int students, int friends, vector<vector<int>>& friendships, int start, vector<bool>& visited, int& timestamp) {
    queue<int> queue;
    
    queue.push(start);
    
    visited[start] = true;
    
    while(!queue.empty()) {
        int size = queue.size();
        for (int i = 0; i < size; i++) {
            int temp = queue.front();
            queue.pop();
        
            for (int neighbor : friendships[temp])
                if (!visited[neighbor]) {
                    visited[neighbor] = true;
                    queue.push(neighbor);
                }
        }
        
        timestamp++;
    }
}

#include <cstdio>
#include <iostream>
#include <vector>
#include <queue>
#include <list>

int bfs(std::vector<std::list<int>> *connections, int n) {
    int *val = (int *) calloc(n, sizeof(int));
    std::queue<std::pair<int, int>> queue;
    queue.push(std::make_pair(0, 0));

    int max = 0;
    std::pair<int, int> s;
    while (!queue.empty()) {
        s = queue.front();
        if (s.second > max) {
            max = s.second;
        }
        queue.pop();
        val[s.first] = 1;
        std::list<int> friends = connections->at(s.first);

        for (auto const &i : friends) {
            if (!val[i]) {
                queue.push(std::make_pair(i, s.second + 1));
                val[i] = -1;

            }
        }

    }

    free(val);
    return max;
}

int main() {
    
    int n, m;
    scanf("%d %d", &n, &m);
    std::vector<std::list<int>> *connections = new std::vector<std::list<int>>(n);
    int i, f1, f2;
    for (i = 0; i < m; i++) {
        scanf("%d %d", &f1, &f2);
        connections->at(f1 - 1).emplace_back(f2 - 1);
        connections->at(f2 - 1).emplace_back(f1 - 1);
    }

    int time = bfs(connections, n);
    printf("%d\n", time);
    delete connections;
    return 0;
}
