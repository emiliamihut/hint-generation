import java.io.*;
import java.util.*;
import java.text.*;
import java.math.*;
import java.util.regex.*;

class MyScanner {
    BufferedReader br;
    StringTokenizer st;
 
    public MyScanner() {
        br = new BufferedReader(new InputStreamReader(System.in));
    }
 
    String next() {
        while (st == null || !st.hasMoreElements()) {
            try {
                st = new StringTokenizer(br.readLine());
            } catch (IOException e) {
                e.printStackTrace();
            }
        }
        return st.nextToken();
    }
 
    int nextInt() {
        return Integer.parseInt(next());
    }
 
    long nextLong() {
        return Long.parseLong(next());
    }
 
    double nextDouble() {
        return Double.parseDouble(next());
    }
 
    String nextLine(){
        String str = "";
        try {
            str = br.readLine();
        } catch (IOException e) {
            e.printStackTrace();
        }
        return str;
    }
}

public class Solution {

    public static void main(String[] args) {
        MyScanner myScanner = new MyScanner();
        int n = 0, m = 0;
        n = myScanner.nextInt();
        m = myScanner.nextInt();
        int[][] v = new int[n + 1][m + 1];
        for(int i = 1; i <= n ; i++) {
            for(int j = 1; j <= m ; j++) {
                v[i][j] = 0;
            }   
        }
        
        int a, b;
        for(int i = 1; i <= n ; i++) {
            a = myScanner.nextInt();
            b = myScanner.nextInt();
            v[a][b] = 1;
        }
        
        int timp = 0, nrCun = 1, aflat = 0;
        int[] cunosc = new int[n + 1];
        cunosc[1] = 1;
        for(int i = 1; i <= n && nrCun < n; i++) {
            aflat = 0;
            for(int j = 1; j <= m ; j++) {
                if (v[i][j] == 1 && cunosc[i] == 1) {
                    aflat = 1;
                    cunosc[j] = 1;
                    nrCun++;
                }
            }
            if(aflat == 1)
                timp++;
        }
        System.out.println(timp);
    }
}
