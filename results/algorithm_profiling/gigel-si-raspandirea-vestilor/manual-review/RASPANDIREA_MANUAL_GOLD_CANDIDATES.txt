public class Solution {
    public static int INF = 0x3f3f3f3f;
    public static final int NMAX = 200000;

    public static class Edge {
        public int node;
        public int cost;

        Edge(int _node, int _cost) {
            node = _node;
            cost = _cost;
        }
    }

    @SuppressWarnings("unchecked")
    static
    ArrayList<Edge>[] adj = new ArrayList[NMAX];

    public static void main(String[] args) {
        
        MyScanner sc = new MyScanner();
        int n = sc.nextInt();
        int m = sc.nextInt();
        for (int i = 1; i <= n; i++) {
            adj[i] = new ArrayList<>();
        }
        for (int i = 1; i <= m; i++) {
            int x, y, w;
            x = sc.nextInt();
            y = sc.nextInt();
            w = 1;
            adj[x].add(new Edge(y, w));
            adj[y].add(new Edge(x, w));
        }

        ArrayList<Integer> results = getResult(1, n);
        
        Integer max = -1;
        for (int i = 1; i <= n; i++) {
            Integer j = results.get(i);
            if (j > max && j != INF) {
                max = j;
            }
        }
        System.out.print(max);
    }

    static class DijkstraNode implements Comparable<DijkstraNode> {
        int dist;
        int node;

        public DijkstraNode(int _dist, int _node) {
            dist = _dist;
            node = _node;
        }

        public int compareTo(DijkstraNode other) {
            if (dist == other.dist) return Integer.compare(node, other.node);
            return Integer.compare(dist, other.dist);
        }
    }

    private static ArrayList<Integer> getResult(int source, int n) {
        
        
        
        
        
        
        
        
        
        

        ArrayList<Integer> d = new ArrayList<>();
        for (int i = 0; i <= n; i++)
            d.add(INF);

        
        
        
        
        TreeSet<DijkstraNode> q = new TreeSet<>();
        d.set(source, 0);
        q.add(new DijkstraNode(d.get(source), source));

        while (!q.isEmpty()) {
            DijkstraNode dnode = q.pollFirst();
            int x = dnode.node;

            for (Edge e : adj[x]) {
                int y = e.node;
                int c = e.cost;

                
                if (d.get(y) > d.get(x) + c) {
                    
                    
                    q.remove(new DijkstraNode(d.get(y), y));
                    
                    d.set(y, d.get(x) + c);
                    q.add(new DijkstraNode(d.get(y), y));
                }
            }
        }

        
        for (int i = 1; i <= n; i++) {
            if (d.get(i) == INF)
                d.set(i, -1);
        }

        return d;
    }
