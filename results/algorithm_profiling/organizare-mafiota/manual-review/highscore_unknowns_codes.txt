    public void createGraph(int N) {
        this.N = N;
        graph = new ArrayList[N+1];
        in = new int[N+1];
        for (int i = 0; i <= N; i++) {
            graph[i] = new ArrayList<>();
            in[i] = 0;
        }
        topSort = new ArrayList<>();

        graph[u].add(v);
        in[v]++;
    }

    public void RoyFloyd() {
        dist = new int[N+1][N+1];
        for (int i = 1; i <= N; i++) {
            for (int j = 1; j <= N; j++) {
                if (i != j)
                    dist[i][j] = Integer.MAX_VALUE / 2;
            }
            for (Integer val : graph[i]) {
                dist[i][val] = 1;
            }
        }
        for (int k = 1; k <= N; k++) {
            for (int i = 1; i <= N; i++) {
                for (int j = 1; j <= N; j++) {
                    if (dist[i][j] > dist[i][k] + dist[k][j]) {
                        dist[i][j] = dist[i][k] + dist[k][j];
                    }
                }
            }
        }
    }

    public int solve() {
        RoyFloyd();
        for (int i = 1; i < N; i++) {
            for (int j = i + 1; j <= N; j++) {
                if (dist[i][j] == Integer.MAX_VALUE / 2 && dist[j][i] == Integer.MAX_VALUE / 2) {
                    return 0;
                }
            }
        }
        return 1;
    }

            vector<int> d(n + 1, -1);
            vector<string> color(n + 1, "alb");

            d[source] = 0;
            queue<int> q;
            q.push(source);

            while (!q.empty()) {
                int u = q.front();
                for (int v : adj[u]) {
                    if (color[v] == "alb") {
                        d[v] = d[u] + 1;
                        color[v] = "gri";
                        q.push(v);
                    }
                }
                color[u] = "negru";
                q.pop();
            }
            
            vector<int> one_dist;
            for (int j = 1; j <= n; j++) {
                one_dist.push_back(d[j]);
            }
            dist.push_back(one_dist);
        }
        
        for (int l = 0; l < n; l++) {
            for (int j = 0; j < n; j++) {
                if (dist[l][j] == -1 && dist[j][l] == -1) {
                    one_sol = 0;
                    break;
                }
            }  
        }
        cout << one_sol;
        cout << " ";
    }
    
    
    return 0;
}

public class Solution {
    static List<List<Integer>> graph;
    static boolean[] visited;

    static void dfs(int node, boolean[] visited, List<List<Integer>> graph) {
        visited[node] = true;
        for (int neighbor : graph.get(node)) {
            if (!visited[neighbor]) {
                dfs(neighbor, visited, graph);
            }
        }
    }

    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String[] tokens = br.readLine().split(" ");
        int N = Integer.parseInt(tokens[0]);
        int M = Integer.parseInt(tokens[1]);
        int Q = Integer.parseInt(tokens[2]);

        for (int q = 0; q < Q; q++) {
            graph = new ArrayList<>();
            for (int i = 0; i <= N; i++) {
                graph.add(new ArrayList<>());
            }

            for (int i = 0; i < M; i++) {
                String[] edge = br.readLine().split(" ");
                int from = Integer.parseInt(edge[0]);
                int to = Integer.parseInt(edge[1]);
                graph.get(from).add(to);
            }

            List<Set<Integer>> reach = new ArrayList<>();
            for (int i = 0; i <= N; i++) reach.add(new HashSet<>());

            for (int i = 1; i <= N; i++) {
                visited = new boolean[N + 1];
                dfs(i, visited, graph);
                for (int j = 1; j <= N; j++) {
                    if (visited[j]) {
                        reach.get(i).add(j);
                    }
                }
            }

            boolean ok = true;
            outer:
            for (int u = 1; u <= N; u++) {
                for (int v = u + 1; v <= N; v++) {
                    if (!reach.get(u).contains(v) && !reach.get(v).contains(u)) {
                        ok = false;
                        break outer;
                    }
                }
            }

            System.out.print((ok ? "1" : "0") + " ");
        }
    }
}
