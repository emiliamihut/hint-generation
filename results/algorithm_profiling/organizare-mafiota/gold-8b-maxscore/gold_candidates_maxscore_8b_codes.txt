bool solve(int n, const vector<vector<int>>& adj)
{
    int in[n + 1];
    memset(in, 0, sizeof(in));
    
    for(int i = 0;i < adj.size();++i)
        for(int j = 0;j < adj[i].size();++j)
            ++in[adj[i][j]];
    queue<int> q;
    for(int i = 1;i <= n;++i)
        if(!in[i])
            q.push(i);
    if(q.size() > 1)
        return false;
    while(!q.empty())
    {
        int nod = q.front();
        q.pop();
        for(int i = 0;i < adj[nod].size();++i)
        {
            int neigh = adj[nod][i];
            --in[neigh];
            if(!in[neigh])
                q.push(neigh);
        }
        if(q.size() > 1)
            return false;
    }
    return true;
}

public class Solution {

    private int N;
    private ArrayList<Integer> topSort;
    private ArrayList<Integer>[] graph;
    private int[] in;

    public void createGraph(int N) {
        this.N = N;
        graph = new ArrayList[N+1];
        in = new int[N+1];
        for (int i = 0; i <= N; i++) {
            graph[i] = new ArrayList<>();
            in[i] = 0;
        }
        topSort = new ArrayList<>();
    }

    public void insertEdge(int u, int v) {
        graph[u].add(v);
        in[v]++;
    }

    public void KahnAlgorithm() {
        ArrayList<Integer> S = new ArrayList<>();
        for (int i = 1; i <= N; i++) {
            if (in[i] == 0) {
                S.add(i);
            }
        }
        while (!S.isEmpty()) {
            int u = S.get(0);
            S.remove(0);
            topSort.add(u);
            for (Integer v : graph[u]) {
                in[v]--;
                if (in[v] == 0) {
                    S.add(v);
                }
            }
        }
        for (int i = 0; i <= N; i++) {
            if (in[i] > 0) {
                topSort.clear();
                return;
            }
        }
    }

    public int solve() {
        KahnAlgorithm();
        for (int i = 0; i < topSort.size() - 1; i++) {
            if (!graph[topSort.get(i)].contains(topSort.get(i+1))) {
                return 0;
            }
        }
        return 1;
    }
